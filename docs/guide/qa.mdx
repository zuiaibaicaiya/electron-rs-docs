# Q&A

## icon图标

### 修改配置文件

``` ts title="rspress.config.ts"
electronRs({
    ignorePack: true,
    appIcon: './logo.png'
})
```

### 修改配置文件

``` ts title="main.ts"
import path from 'path'
const config: BrowserWindowConstructorOptions = {
  icon: resolve(__dirname, 'logo.png'),
}
mainWindow = new BrowserWindow(config)
```

### 修改package.json

``` json title="package.json"
{
  "build": {
    "icon": "./logo.png",
    "dmg": {
      "icon": "./logo.icns",
      "background": "./logo.png",
      "window": {
        "x": 0,
        "y": 0,
        "width": 512,
        "height": 512
      }
    }
  }
}

```
